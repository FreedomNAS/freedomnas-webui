# What the WebUI needs to start. Three of these are not direct imports and are
# the easiest to forget: jinja2 arrives through fastapi.templating, uvicorn is
# how the supervisor launches the app, and websockets is what lets uvicorn
# serve one. The distribution packages that provide these are named in
# debian/control, PKGBUILD, the Gentoo ebuild and freedomnas-profiles'
# packages/core.json; this file is the list they all answer to.
fastapi
uvicorn
websockets      # uvicorn's WebSocket protocol: without it /ws/updates/terminal is a 404
jinja2          # fastapi.templating, built at import time in api/main.py
python-pam      # api/deps.py: `import pam`. Debian's python3-pam is a
                # different, older library whose module is PAM; the one
                # that provides `pam` there is python3-pampy
PyYAML          # configuration import/export
